    48→    48→function boxMesh(
    49→    49→  w: number, h: number, d: number, mat: THREE.Material,
    50→    50→  x: number, y: number, z: number, ry = 0,
    51→    51→  parent?: THREE.Object3D

    61→    61→export function aabbFromCenter(cx: number, cz: number, w: number, d: number): AABB {
    62→    62→  return { minX: cx - w / 2, maxX: cx + w / 2, minZ: cz - d / 2, maxZ: cz + d / 2 };

    65→    65→export function resolveCollisions(pos: THREE.Vector3, radius: number, colliders: AABB[]) {
    66→    66→  for (const c of colliders) {
    67→    67→    const cx = Math.max(c.minX, Math.min(pos.x, c.maxX));
    68→    68→    const cz = Math.max(c.minZ, Math.min(pos.z, c.maxZ));
    69→    69→    const dx = pos.x - cx, dz = pos.z - cz;
    70→    70→    const d2 = dx * dx + dz * dz;
    71→    71→    if (d2 < radius * radius) {
    72→    72→      if (d2 > 1e-8) {
    73→    73→        const d = Math.sqrt(d2);
    74→    74→        pos.x = cx + (dx / d) * radius;
    75→    75→        pos.z = cz + (dz / d) * radius;
    76→    76→      } else {
    77→    77→        // inside: push out along smallest penetration axis
    78→    78→        const px = Math.min(pos.x - c.minX, c.maxX - pos.x);
    79→    79→        const pz = Math.min(pos.z - c.minZ, c.maxZ - pos.z);
    80→    80→        if (px < pz) pos.x = (pos.x - c.minX < c.maxX - pos.x) ? c.minX - radius : c.maxX + radius;
    81→    81→        else pos.z = (pos.z - c.minZ < c.maxZ - pos.z) ? c.minZ - radius : c.maxZ + radius;
    82→    82→      }
    83→    83→    }
    84→    84→  }
    85→    85→}

    88→    88→export function raycastClear(from: THREE.Vector3, to: THREE.Vector3, occluders: THREE.Object3D[]): boolean {
    89→    89→  const dir = new THREE.Vector3().subVectors(to, from);
    90→    90→  const len = dir.length();
    91→    91→  if (len < 0.01) return true;
    92→    92→  dir.normalize();
    93→    93→  _ray.set(from, dir);
    94→    94→  _ray.far = len - 0.1;
    95→    95→  const hits = _ray.intersectObjects(occluders, false);
    96→    96→  return hits.length === 0;
    97→    97→}

   307→   307→  // house with a door gap on one side. N = -z, S = +z, W = -x, E = +x
   308→   308→  interface HouseCfg {
   309→   309→    x: number; z: number; w: number; d: number; h: number;
   310→   310→    door: 'N' | 'S' | 'E' | 'W';
   311→   311→    doorW?: number;
   312→   312→    locked?: string; // door id if locked
   313→   313→    windows?: { side: 'N' | 'S' | 'E' | 'W'; off: number; lit?: boolean }[];
   314→   314→    interior?: [string, number, number, number?][];
   315→   315→    roof?: boolean;
   316→   316→    stone?: boolean;
   317→   317→  }
   318→   318→  const makeHouse = (cfg: HouseCfg) => {
   319→   319→    const { x, z, w, d, h } = cfg;
   320→   320→    const mat = cfg.stone ? matStone : matPlank;
   321→   321→    const dw = cfg.doorW ?? 1.4;
   322→   322→    const wallSegs = (side: 'N' | 'S' | 'E' | 'W', winOffs: number[]) => {
   323→   323→      const horiz = side === 'N' || side === 'S';
   324→   324→      const fixed = side === 'N' ? z - d / 2 : side === 'S' ? z + d / 2 : side === 'W' ? x - w / 2 : x + w / 2;
   325→   325→      const len = horiz ? w : d;
   326→   326→      const isDoorSide = side === cfg.door;
   327→   327→      const segments: [number, number][] = isDoorSide
   328→   328→        ? [[-len / 2, -dw / 2], [dw / 2, len / 2]]
   329→   329→        : [[-len / 2, len / 2]];
   330→   330→      for (const [a, b] of segments) {
   331→   331→        const segLen = b - a;
   332→   332→        if (segLen < 0.05) continue;
   333→   333→        const mid = (a + b) / 2;
   334→   334→        if (horiz) wall(x + mid, h / 2, fixed, segLen, h, 0.3, mat);
   335→   335→        else wall(fixed, h / 2, z + mid, 0.3, h, segLen, mat);
   336→   336→      }
   337→   337→      // lintel above door
   338→   338→      if (isDoorSide) {
   339→   339→        if (horiz) wall(x, h - 0.35, fixed, dw, 0.7, 0.3, mat, false);
   340→   340→        else wall(fixed, h - 0.35, z, 0.3, 0.7, dw, mat, false);
   341→   341→      }
   342→   342→      for (const off of winOffs) {
   343→   343→        if (isDoorSide && Math.abs(off) < 2) continue;
