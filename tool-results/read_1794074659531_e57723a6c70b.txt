     4→/* ---------------- math utils ---------------- */
     5→export const clamp = (v: number, a: number, b: number) => Math.min(b, Math.max(a, v));
     6→export const lerp = (a: number, b: number, t: number) => a + (b - a) * t;
     7→export const dist2D = (ax: number, az: number, bx: number, bz: number) => Math.hypot(ax - bx, az - bz);
     8→export function mulberry32(seed: number) {
     9→  let a = seed >>> 0;
    10→  return function () {
    11→    a |= 0; a = (a + 0x6D2B79F5) | 0;
    12→    let t = Math.imul(a ^ (a >>> 15), 1 | a);
    13→    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
    14→    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
    15→  };
    16→}

    83→function noiseCanvas(size: number, base: string, variants: string[], grain: number, seed = 1): HTMLCanvasElement {
    84→  const c = document.createElement('canvas');
    85→  c.width = c.height = size;
    86→  const g = c.getContext('2d')!;
    87→  const rnd = mulberry32(seed);
    88→  g.fillStyle = base;
    89→  g.fillRect(0, 0, size, size);
    90→  for (let i = 0; i < variants.length; i++) {
    91→    g.fillStyle = variants[i];
    92→    for (let n = 0; n < size * size * 0.02; n++) {
    93→      const x = rnd() * size, y = rnd() * size, r = 1 + rnd() * (size / 32);
    94→      g.globalAlpha = 0.12 + rnd() * 0.22;
    95→      g.beginPath(); g.arc(x, y, r, 0, Math.PI * 2); g.fill();
    96→    }
    97→  }
    98→  g.globalAlpha = 1;
    99→  const img = g.getImageData(0, 0, size, size);
   100→  for (let i = 0; i < img.data.length; i += 4) {
   101→    const n = (rnd() - 0.5) * grain;
   102→    img.data[i] = clamp(img.data[i] + n, 0, 255);
   103→    img.data[i + 1] = clamp(img.data[i + 1] + n, 0, 255);
   104→    img.data[i + 2] = clamp(img.data[i + 2] + n, 0, 255);
   105→  }
   106→  g.putImageData(img, 0, 0);
   107→  return c;
   108→}

   155→      for (let y = 0; y < S; y += 40) { for (let x = ((y / 40) % 2) * 20; x < S; x += 40) { g2.beginPath(); g2.moveTo(x, y); g2.lineTo(x, y + 20); g2.stroke(); } }
   156→      break;
   157→    }

   192→/* ---------------- audio engine (fully procedural) ---------------- */
   193→export class AudioEngine {
   194→  ctx: AudioContext | null = null;
   195→  master: GainNode | null = null;
   196→  private windGain: GainNode | null = null;
   197→  private droneOsc: OscillatorNode | null = null;
   198→  private droneGain: GainNode | null = null;
   199→  private dissGain: GainNode | null = null;
   200→  private growlGain: GainNode | null = null;
   201→  private growlOsc: OscillatorNode | null = null;
   202→  private heartTimer = 0;
   203→  danger = 0; // 0..1 drives heartbeat & tension layers
   204→  volume = 0.8;
   205→
   206→  ensure() {
   207→    if (this.ctx) { if (this.ctx.state === 'suspended') void this.ctx.resume(); return; }
   208→    const AC = window.AudioContext || (window as unknown as { webkitAudioContext: typeof AudioContext }).webkitAudioContext;
   209→    this.ctx = new AC();
   210→    this.master = this.ctx.createGain();
   211→    this.master.gain.value = this.volume;
   212→    this.master.connect(this.ctx.destination);
   213→    this.startAmbience();
   214→  }
   215→
   216→  setVolume(v: number) {
   217→    this.volume = v;
   218→    if (this.master) this.master.gain.value = v;
   219→  }
   220→
   221→  private noiseBuffer(seconds = 2): AudioBuffer {
   222→    const ctx = this.ctx!;
   223→    const buf = ctx.createBuffer(1, Math.max(1, Math.floor(ctx.sampleRate * seconds)), ctx.sampleRate);
   224→    const d = buf.getChannelData(0);
   225→    let last = 0;
   226→    for (let i = 0; i < d.length; i++) {
   227→      const w = Math.random() * 2 - 1;
   228→      last = (last + 0.02 * w) / 1.02;
   229→      d[i] = last * 3.2;
   230→    }
   231→    return buf;
   232→  }
   233→
   234→  private startAmbience() {
   235→    const ctx = this.ctx!;
   236→    // wind
   237→    const src = ctx.createBufferSource();
   238→    src.buffer = this.noiseBuffer(4); src.loop = true;
   239→    const lp = ctx.createBiquadFilter(); lp.type = 'lowpass'; lp.frequency.value = 420; lp.Q.value = 0.6;
   240→    this.windGain = ctx.createGain(); this.windGain.gain.value = 0.14;
   241→    const lfo = ctx.createOscillator(); lfo.frequency.value = 0.09;
   242→    const lfoG = ctx.createGain(); lfoG.gain.value = 0.05;
   243→    lfo.connect(lfoG); lfoG.connect(this.windGain.gain);
   244→    src.connect(lp); lp.connect(this.windGain); this.windGain.connect(this.master!);
   245→    src.start(); lfo.start();
   246→    // dread drone
   247→    this.droneOsc = ctx.createOscillator(); this.droneOsc.type = 'sine'; this.droneOsc.frequency.value = 46;
   248→    const dr2 = ctx.createOscillator(); dr2.type = 'sine'; dr2.frequency.value = 46.7;
   249→    this.droneGain = ctx.createGain(); this.droneGain.gain.value = 0.05;
   250→    this.dissGain = ctx.createGain(); this.dissGain.gain.value = 0.0;
   251→    const diss = ctx.createOscillator(); diss.type = 'sawtooth'; diss.frequency.value = 92;
   252→    const dissLp = ctx.createBiquadFilter(); dissLp.type = 'lowpass'; dissLp.frequency.value = 300;
   253→    this.droneOsc.connect(this.droneGain); dr2.connect(this.droneGain);
   254→    this.droneGain.connect(this.master!);
   255→    diss.connect(dissLp); dissLp.connect(this.dissGain); this.dissGain.connect(this.master!);
   256→    this.droneOsc.start(); dr2.start(); diss.start();
   257→    // monster growl loop (gain driven per-frame)
   258→    this.growlOsc = ctx.createOscillator(); this.growlOsc.type = 'sawtooth'; this.growlOsc.frequency.value = 58;
   259→    const growlLp = ctx.createBiquadFilter(); growlLp.type = 'lowpass'; growlLp.frequency.value = 190;
   260→    const growlLfo = ctx.createOscillator(); growlLfo.frequency.value = 5.2;
   261→    const growlLfoG = ctx.createGain(); growlLfoG.gain.value = 14;
   262→    growlLfo.connect(growlLfoG); growlLfoG.connect(this.growlOsc.frequency);
   263→    this.growlGain = ctx.createGain(); this.growlGain.gain.value = 0;
   264→    this.growlOsc.connect(growlLp); growlLp.connect(this.growlGain); this.growlGain.connect(this.master!);
   265→    this.growlOsc.start(); growlLfo.start();
   266→  }
   267→
   268→  update(dt: number) {
   269→    if (!this.ctx || !this.droneGain || !this.droneOsc || !this.dissGain) return;
   270→    const t = this.ctx.currentTime;
   271→    this.droneGain.gain.setTargetAtTime(0.04 + this.danger * 0.09, t, 0.4);
   272→    this.droneOsc.frequency.setTargetAtTime(46 + this.danger * 10, t, 0.6);
   273→    this.dissGain.gain.setTargetAtTime(this.danger * 0.045, t, 0.5);
   274→    if (this.windGain) this.windGain.gain.setTargetAtTime(0.10 + this.danger * 0.08, t, 1.2);
   275→    // heartbeat
   276→    if (this.danger > 0.25) {
   277→      this.heartTimer -= dt;
   278→      if (this.heartTimer <= 0) {
   279→        const bpm = 42 + this.danger * 78;
   280→        this.heartTimer = 60 / bpm;
   281→        this.thump(0.5 + this.danger * 0.5);
   282→        setTimeout(() => this.thump(0.35 + this.danger * 0.35), 150);
   283→      }
   284→    }
   285→  }
   286→
   287→  private thump(vol: number) {
   288→    const ctx = this.ctx; if (!ctx || !this.master) return;
   289→    const o = ctx.createOscillator(); o.type = 'sine';
   290→    o.frequency.setValueAtTime(58, ctx.currentTime);
   291→    o.frequency.exponentialRampToValueAtTime(30, ctx.currentTime + 0.12);
   292→    const g = ctx.createGain();
   293→    g.gain.setValueAtTime(vol * 0.5, ctx.currentTime);
   294→    g.gain.exponentialRampToValueAtTime(0.001, ctx.currentTime + 0.22);
   295→    o.connect(g); g.connect(this.master);
   296→    o.start(); o.stop(ctx.currentTime + 0.25);
   297→  }

   299→  private env(node: AudioNode, vol: number, dur: number, attack = 0.005): GainNode {
   300→    const ctx = this.ctx!;
   301→    const g = ctx.createGain();
   302→    g.gain.setValueAtTime(0.0001, ctx.currentTime);
   303→    g.gain.linearRampToValueAtTime(vol, ctx.currentTime + attack);
   304→    g.gain.exponentialRampToValueAtTime(0.0001, ctx.currentTime + dur);
   305→    node.connect(g); g.connect(this.master!);
   306→    return g;
   307→  }

   309→  // spatial helper: returns [gain, pan]
   310→  spatial(listener: THREE.Vector3 | null, listenerYaw: number, pos: THREE.Vector3 | null, base: number): [number, number] {
   311→    if (!listener || !pos) return [base, 0];
   312→    const dx = pos.x - listener.x, dz = pos.z - listener.z;
   313→    const d = Math.hypot(dx, dz);
   314→    const gain = base * clamp(1 - d / 38, 0, 1) ** 1.4;
   315→    const ang = Math.atan2(dx, dz) - listenerYaw;
   316→    const pan = clamp(Math.sin(ang), -1, 1) * clamp(1 - d / 40, 0, 1);
   317→    return [gain, pan];
   318→  }

   320→  play3d(name: string, listener: THREE.Vector3 | null, listenerYaw: number, pos: THREE.Vector3 | null, base = 0.5) {
   321→    const [gain, pan] = this.spatial(listener, listenerYaw, pos, base);
   322→    if (gain < 0.004) return;
   323→    this.play(name, gain, pan);
   324→  }

   326→  play(name: string, vol = 0.5, pan = 0) {
   327→    const ctx = this.ctx; if (!ctx || !this.master) return;
   328→    const out = ctx.createStereoPanner(); out.pan.value = clamp(pan, -1, 1); out.connect(this.master);
   329→    const t = ctx.currentTime;
   330→    switch (name) {
   331→      case 'step': {
   332→        const s = ctx.createBufferSource(); s.buffer = this.noiseBuffer(0.1);
   333→        const f = ctx.createBiquadFilter(); f.type = 'bandpass'; f.frequency.value = 240 + Math.random() * 160; f.Q.value = 1.1;
   334→        s.connect(f);
   335→        const g = ctx.createGain(); g.gain.setValueAtTime(vol * 0.5, t); g.gain.exponentialRampToValueAtTime(0.0001, t + 0.09);
   336→        f.connect(g); g.connect(out); s.start(); s.stop(t + 0.1);
   337→        break;
   338→      }
   339→      case 'shot': {
   340→        const s = ctx.createBufferSource(); s.buffer = this.noiseBuffer(0.4);
   341→        const f = ctx.createBiquadFilter(); f.type = 'lowpass';
   342→        f.frequency.setValueAtTime(3200, t); f.frequency.exponentialRampToValueAtTime(180, t + 0.28);
   343→        s.connect(f);
   344→        const g = ctx.createGain(); g.gain.setValueAtTime(vol, t); g.gain.exponentialRampToValueAtTime(0.0001, t + 0.32);
   345→        f.connect(g); g.connect(out);
   346→        const o = ctx.createOscillator(); o.type = 'triangle';
   347→        o.frequency.setValueAtTime(110, t); o.frequency.exponentialRampToValueAtTime(38, t + 0.16);
   348→        const g2 = ctx.createGain(); g2.gain.setValueAtTime(vol * 0.9, t); g2.gain.exponentialRampToValueAtTime(0.0001, t + 0.2);
   349→        o.connect(g2); g2.connect(out);
   350→        s.start(); s.stop(t + 0.4); o.start(); o.stop(t + 0.22);
   351→        break;
   352→      }
   353→      case 'dry': { this.blip(out, 1400, 0.04, vol * 0.4, 'square'); break; }
   354→      case 'reload': { this.blip(out, 900, 0.05, vol * 0.5, 'square'); setTimeout(() => this.blip(out, 620, 0.06, vol * 0.5, 'square'), 130); setTimeout(() => this.blip(out, 1100, 0.05, vol * 0.6, 'square'), 300); break; }
   355→      case 'knife': {
   356→        const s = ctx.createBufferSource(); s.buffer = this.noiseBuffer(0.2);
   357→        const f = ctx.createBiquadFilter(); f.type = 'bandpass'; f.Q.value = 2.5;
   358→        f.frequency.setValueAtTime(600, t); f.frequency.exponentialRampToValueAtTime(3600, t + 0.12);
   359→        s.connect(f);
   360→        const g = ctx.createGain(); g.gain.setValueAtTime(vol * 0.7, t); g.gain.exponentialRampToValueAtTime(0.0001, t + 0.15);
   361→        f.connect(g); g.connect(out); s.start(); s.stop(t + 0.2);
   362→        break;
   363→      }
   364→      case 'flesh': {
   365→        const s = ctx.createBufferSource(); s.buffer = this.noiseBuffer(0.25);
   366→        const f = ctx.createBiquadFilter(); f.type = 'lowpass'; f.frequency.value = 380;
   367→        s.connect(f);
   368→        const g = ctx.createGain(); g.gain.setValueAtTime(vol * 0.9, t); g.gain.exponentialRampToValueAtTime(0.0001, t + 0.22);
   369→        f.connect(g); g.connect(out); s.start(); s.stop(t + 0.25);
   370→        break;
   371→      }
   372→      case 'pickup': { this.tone(out, 520, 0.09, vol * 0.35, 'sine'); setTimeout(() => this.tone(out, 780, 0.12, vol * 0.35, 'sine'), 80); break; }
   373→      case 'paper': {
   374→        const s = ctx.createBufferSource(); s.buffer = this.noiseBuffer(0.3);
   375→        const f = ctx.createBiquadFilter(); f.type = 'highpass'; f.frequency.value = 1800;
   376→        s.connect(f);
   377→        const g = ctx.createGain(); g.gain.setValueAtTime(vol * 0.35, t); g.gain.exponentialRampToValueAtTime(0.0001, t + 0.28);
   378→        f.connect(g); g.connect(out); s.start(); s.stop(t + 0.3);
   379→        break;
   380→      }
   381→      case 'door': {
   382→        const o = ctx.createOscillator(); o.type = 'sawtooth';
   383→        o.frequency.setValueAtTime(85, t); o.frequency.exponentialRampToValueAtTime(55, t + 0.7);
   384→        const f = ctx.createBiquadFilter(); f.type = 'bandpass'; f.frequency.value = 300; f.Q.value = 4;
   385→        o.connect(f);
   386→        const g = ctx.createGain(); g.gain.setValueAtTime(vol * 0.45, t); g.gain.exponentialRampToValueAtTime(0.0001, t + 0.75);
   387→        f.connect(g); g.connect(out); o.start(); o.stop(t + 0.8);
   388→        break;
   389→      }
   390→      case 'brazier': {
   391→        const s = ctx.createBufferSource(); s.buffer = this.noiseBuffer(1.2);
   392→        const f = ctx.createBiquadFilter(); f.type = 'lowpass';

   478→  private blip(out: AudioNode, freq: number, dur: number, vol: number, type: OscillatorType) {
   479→    const ctx = this.ctx!; const t = ctx.currentTime;

   483→  }

   495→/* keyboard normalization: fall back to e.key when e.code is empty (some webviews) */
   496→const KEY_MAP: Record<string, string> = {
   497→  w: 'KeyW', a: 'KeyA', s: 'KeyS', d: 'KeyD', e: 'KeyE', r: 'KeyR', f: 'KeyF', q: 'KeyQ',
   498→  h: 'KeyH', v: 'KeyV', c: 'KeyC', i: 'KeyI', escape: 'Escape', shift: 'ShiftLeft',
   499→  control: 'ControlLeft', '1': 'Digit1', '2': 'Digit2', '3': 'Digit3', ' ': 'Space',
   500→  arrowup: 'ArrowUp', arrowdown: 'ArrowDown', arrowleft: 'ArrowLeft', arrowright: 'ArrowRight',
   501→};
   502→function normalizeKey(e: KeyboardEvent): string {
   503→  if (e.code) return e.code;
   504→  const k = e.key.toLowerCase();
   505→  return KEY_MAP[k] || 'Key_' + k.toUpperCase();
   506→}

   508→/* ---------------- input ---------------- */
   509→export class Input {
   510→  keys: Record<string, boolean> = {};
   511→  buttons: Record<string, boolean> = {}; // unified action buttons (touch + kb)
   512→  look = { x: 0, y: 0 }; // consumed each frame
   513→  touchMove = { x: 0, y: 0 };
   514→  pointerLocked = false;
   515→  private el: HTMLElement | null = null;
   516→  onKeyExtra: ((code: string) => void) | null = null;
   517→
   518→  attach(el: HTMLElement) {
   519→    this.el = el;
   520→    window.addEventListener('keydown', this.kd);
   521→    window.addEventListener('keyup', this.ku);
   522→    el.addEventListener('mousedown', this.md);
   523→    window.addEventListener('mouseup', this.mu);
   524→    window.addEventListener('mousemove', this.mm);
   525→    document.addEventListener('pointerlockchange', this.plc);
   526→    el.addEventListener('contextmenu', (e) => e.preventDefault());
   527→  }
   528→  detach() {
   529→    window.removeEventListener('keydown', this.kd);
   530→    window.removeEventListener('keyup', this.ku);
   531→    window.removeEventListener('mouseup', this.mu);
   532→    window.removeEventListener('mousemove', this.mm);
   533→    document.removeEventListener('pointerlockchange', this.plc);
   534→  }
   535→  private kd = (e: KeyboardEvent) => {
   536→    if (e.repeat) { return; }
   537→    const code = normalizeKey(e);
   538→    this.keys[code] = true;
   539→    if (this.onKeyExtra) this.onKeyExtra(code);
   540→    if (['Space', 'Tab'].includes(code)) e.preventDefault();
   541→  };
   542→  private ku = (e: KeyboardEvent) => { this.keys[normalizeKey(e)] = false; };
   543→  private md = (e: MouseEvent) => {
   544→    if (!this.pointerLocked && this.el) { void this.el.requestPointerLock?.(); }
   545→    if (e.button === 0) this.buttons['attack'] = true;
   546→    if (e.button === 2) this.buttons['aim'] = true;
   547→  };
   548→  private mu = (e: MouseEvent) => {
   549→    if (e.button === 0) this.buttons['attack'] = false;
   550→    if (e.button === 2) this.buttons['aim'] = false;
   551→  };
   552→  private mm = (e: MouseEvent) => {
   553→    if (!this.pointerLocked) return;
   554→    this.look.x += e.movementX * 0.0022;
   555→    this.look.y += e.movementY * 0.0022;
   556→  };
   557→  private plc = () => { this.pointerLocked = document.pointerLockElement === this.el; };
   558→
   559→  requestLock() { if (this.el && !this.pointerLocked) void this.el.requestPointerLock?.(); }
   560→  exitLock() { if (document.pointerLockElement) document.exitPointerLock(); }
   561→
   562→  // touch hooks (called from ui.ts)
   563→  touchLook(dx: number, dy: number) { this.look.x += dx * 0.0045; this.look.y += dy * 0.0045; }
   564→  touchMoveVec(x: number, y: number) { this.touchMove.x = x; this.touchMove.y = y; }
   565→  setButton(name: string, down: boolean) { this.buttons[name] = down; if (down && this.onKeyExtra) this.onKeyExtra('T_' + name); }
   566→
   567→  consumeLook(): { x: number, y: number } {
   568→    const r = { x: this.look.x, y: this.look.y };
   569→    this.look.x = 0; this.look.y = 0;
   570→    return r;
   571→  }
   572→  moveVector(): { x: number, y: number } {
   573→    let x = 0, y = 0;
   574→    if (this.keys['KeyW'] || this.keys['ArrowUp']) y -= 1;
   575→    if (this.keys['KeyS'] || this.keys['ArrowDown']) y += 1;
   576→    if (this.keys['KeyA'] || this.keys['ArrowLeft']) x -= 1;
   577→    if (this.keys['KeyD'] || this.keys['ArrowRight']) x += 1;
   578→    const len = Math.hypot(x, y);
   579→    if (len > 1) { x /= len; y /= len; }
   580→    if (Math.hypot(this.touchMove.x, this.touchMove.y) > 0.12) { x = this.touchMove.x; y = this.touchMove.y; }
   581→    return { x, y };
   582→  }
   583→  down(code: string) { return !!this.keys[code] || !!this.buttons[code]; }
   584→}
