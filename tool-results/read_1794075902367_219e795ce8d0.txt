     4→/* ---------------- math utils ---------------- */
     5→export const clamp = (v: number, a: number, b: number) => Math.min(b, Math.max(a, v));
     6→export const lerp = (a: number, b: number, t: number) => a + (b - a) * t;
     7→export const dist2D = (ax: number, az: number, bx: number, bz: number) => Math.hypot(ax - bx, az - bz);
     8→export function mulberry32(seed: number) {
     9→  let a = seed >>> 0;
    10→  return function () {
    11→    a |= 0; a = (a + 0x6D2B79F5) | 0;
    12→    let t = Math.imul(a ^ (a >>> 15), 1 | a);
    13→    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
    14→    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
    15→  };
    16→}

    91→function noiseCanvas(size: number, base: string, variants: string[], grain: number, seed = 1): HTMLCanvasElement {
    92→  const c = document.createElement('canvas');
    93→  c.width = c.height = size;
    94→  const g = c.getContext('2d')!;
    95→  const rnd = mulberry32(seed);
    96→  g.fillStyle = base;
    97→  g.fillRect(0, 0, size, size);
    98→  for (let i = 0; i < variants.length; i++) {
    99→    g.fillStyle = variants[i];
   100→    for (let n = 0; n < size * size * 0.02; n++) {
   101→      const x = rnd() * size, y = rnd() * size, r = 1 + rnd() * (size / 32);
   102→      g.globalAlpha = 0.12 + rnd() * 0.22;
   103→      g.beginPath(); g.arc(x, y, r, 0, Math.PI * 2); g.fill();
   104→    }
   105→  }
   106→  g.globalAlpha = 1;
   107→  const img = g.getImageData(0, 0, size, size);
   108→  for (let i = 0; i < img.data.length; i += 4) {
   109→    const n = (rnd() - 0.5) * grain;
   110→    img.data[i] = clamp(img.data[i] + n, 0, 255);
   111→    img.data[i + 1] = clamp(img.data[i + 1] + n, 0, 255);
   112→    img.data[i + 2] = clamp(img.data[i + 2] + n, 0, 255);
   113→  }
   114→  g.putImageData(img, 0, 0);
   115→  return c;
   116→}

   258→/* soft radial sprite used by fog planes, glow, smoke, lightning dust */
   259→export function getSpriteTexture(kind: 'puff' | 'flame' | 'glow' | 'blood' | 'hole' | 'rain'): THREE.Texture {
   260→  const key = 'sprite_' + kind;
   261→  const cached = texCache.get(key);
   262→  if (cached) return cached;
   263→  const S = 128;
   264→  const c = document.createElement('canvas');
   265→  c.width = c.height = S;
   266→  const g = c.getContext('2d')!;
   267→  if (kind === 'puff' || kind === 'glow' || kind === 'flame') {
   268→    const grad = g.createRadialGradient(S / 2, S / 2, 2, S / 2, S / 2, S / 2);
   269→    if (kind === 'puff') { grad.addColorStop(0, 'rgba(180,185,200,0.55)'); grad.addColorStop(0.5, 'rgba(150,155,175,0.22)'); grad.addColorStop(1, 'rgba(140,145,165,0)'); }
   270→    else if (kind === 'glow') { grad.addColorStop(0, 'rgba(255,214,150,0.9)'); grad.addColorStop(0.4, 'rgba(255,170,80,0.35)'); grad.addColorStop(1, 'rgba(255,150,50,0)'); }
   271→    else { grad.addColorStop(0, 'rgba(255,235,190,0.95)'); grad.addColorStop(0.35, 'rgba(255,160,60,0.8)'); grad.addColorStop(0.7, 'rgba(200,70,20,0.35)'); grad.addColorStop(1, 'rgba(120,30,10,0)'); }
   272→    g.fillStyle = grad;
   273→    g.fillRect(0, 0, S, S);
   274→  } else if (kind === 'blood') {
   275→    const rnd2 = mulberry32(31);
   276→    g.fillStyle = 'rgba(90,8,8,0.9)';
   277→    g.beginPath(); g.arc(S / 2, S / 2, S * 0.22, 0, Math.PI * 2); g.fill();
   278→    for (let i = 0; i < 26; i++) {
   279→      const a = rnd2() * Math.PI * 2, d = S * (0.16 + rnd2() * 0.3);
   280→      g.beginPath(); g.arc(S / 2 + Math.cos(a) * d, S / 2 + Math.sin(a) * d, 1.5 + rnd2() * 5, 0, Math.PI * 2);
   281→      g.fillStyle = `rgba(${70 + rnd2() * 40},6,6,${0.5 + rnd2() * 0.45})`; g.fill();
   282→    }
   283→  } else if (kind === 'hole') {
   284→    const grad = g.createRadialGradient(S / 2, S / 2, 1, S / 2, S / 2, S / 2);
   285→    grad.addColorStop(0, 'rgba(8,8,8,0.95)'); grad.addColorStop(0.35, 'rgba(20,18,16,0.8)'); grad.addColorStop(0.6, 'rgba(40,36,30,0.35)'); grad.addColorStop(1, 'rgba(40,36,30,0)');
   286→    g.fillStyle = grad; g.fillRect(0, 0, S, S);
   287→  } else {
   288→    // rain streak
   289→    const grad = g.createLinearGradient(0, 0, 0, S);
   290→    grad.addColorStop(0, 'rgba(160,180,210,0)'); grad.addColorStop(0.5, 'rgba(170,190,220,0.5)'); grad.addColorStop(1, 'rgba(160,180,210,0)');
   291→    g.fillStyle = grad; g.fillRect(S / 2 - 1, 0, 2, S);
   292→  }
   293→  const t = new THREE.CanvasTexture(c);
   294→  t.colorSpace = THREE.SRGBColorSpace;
   295→  texCache.set(key, t);
   296→  return t;
   297→}

   306→/* ---------------- audio engine (fully procedural) ---------------- */
   307→export class AudioEngine {
   308→  ctx: AudioContext | null = null;
   309→  master: GainNode | null = null;
   310→  private windGain: GainNode | null = null;
   311→  private droneOsc: OscillatorNode | null = null;
   312→  private droneGain: GainNode | null = null;
   313→  private dissGain: GainNode | null = null;
   314→  private growlGain: GainNode | null = null;
   315→  private growlOsc: OscillatorNode | null = null;
   316→  private heartTimer = 0;
   317→  danger = 0; // 0..1 drives heartbeat & tension layers
   318→  volume = 0.8;
   319→  private rainGain: GainNode | null = null;
   320→  private chaseGain: GainNode | null = null;
   321→  private chaseOsc: OscillatorNode | null = null;
   322→
   323→  /** 0..1 — intense chase music layer (pounding strings + pulse) */
   324→  setChase(x: number) {
   325→    if (!this.chaseGain || !this.ctx) return;
   326→    this.chaseGain.gain.setTargetAtTime(clamp(x, 0, 1) * 0.16, this.ctx.currentTime, 0.35);
   327→    if (this.chaseOsc) this.chaseOsc.frequency.setTargetAtTime(98 + x * 40, this.ctx.currentTime, 0.5);
   328→  }
   329→
   330→  /** 0..1 — rain ambience loop */
   331→  setRain(x: number) {
   332→    if (!this.rainGain || !this.ctx) return;
   333→    this.rainGain.gain.setTargetAtTime(clamp(x, 0, 1) * 0.16, this.ctx.currentTime, 1.2);
   334→  }
   335→
   336→  ensure() {
   337→    if (this.ctx) { if (this.ctx.state === 'suspended') void this.ctx.resume(); return; }
   338→    const AC = window.AudioContext || (window as unknown as { webkitAudioContext: typeof AudioContext }).webkitAudioContext;
   339→    this.ctx = new AC();
   340→    this.master = this.ctx.createGain();
   341→    this.master.gain.value = this.volume;
   342→    this.master.connect(this.ctx.destination);
   343→    this.startAmbience();
   344→  }
   345→
   346→  setVolume(v: number) {
   347→    this.volume = v;
   348→    if (this.master) this.master.gain.value = v;
   349→  }
   350→
   351→  private noiseBuffer(seconds = 2): AudioBuffer {
   352→    const ctx = this.ctx!;
   353→    const buf = ctx.createBuffer(1, Math.max(1, Math.floor(ctx.sampleRate * seconds)), ctx.sampleRate);
   354→    const d = buf.getChannelData(0);
   355→    let last = 0;
   356→    for (let i = 0; i < d.length; i++) {
   357→      const w = Math.random() * 2 - 1;
   358→      last = (last + 0.02 * w) / 1.02;
   359→      d[i] = last * 3.2;
   360→    }
   361→    return buf;
   362→  }
   363→
   364→  private startAmbience() {
   365→    const ctx = this.ctx!;
   366→    // wind
   367→    const src = ctx.createBufferSource();
   368→    src.buffer = this.noiseBuffer(4); src.loop = true;
   369→    const lp = ctx.createBiquadFilter(); lp.type = 'lowpass'; lp.frequency.value = 420; lp.Q.value = 0.6;
   370→    this.windGain = ctx.createGain(); this.windGain.gain.value = 0.14;
   371→    const lfo = ctx.createOscillator(); lfo.frequency.value = 0.09;
   372→    const lfoG = ctx.createGain(); lfoG.gain.value = 0.05;
   373→    lfo.connect(lfoG); lfoG.connect(this.windGain.gain);
   374→    src.connect(lp); lp.connect(this.windGain); this.windGain.connect(this.master!);
   375→    src.start(); lfo.start();
   376→    // dread drone
   377→    this.droneOsc = ctx.createOscillator(); this.droneOsc.type = 'sine'; this.droneOsc.frequency.value = 46;
   378→    const dr2 = ctx.createOscillator(); dr2.type = 'sine'; dr2.frequency.value = 46.7;
   379→    this.droneGain = ctx.createGain(); this.droneGain.gain.value = 0.05;
   380→    this.dissGain = ctx.createGain(); this.dissGain.gain.value = 0.0;
   381→    const diss = ctx.createOscillator(); diss.type = 'sawtooth'; diss.frequency.value = 92;
   382→    const dissLp = ctx.createBiquadFilter(); dissLp.type = 'lowpass'; dissLp.frequency.value = 300;
   383→    this.droneOsc.connect(this.droneGain); dr2.connect(this.droneGain);
   384→    this.droneGain.connect(this.master!);
   385→    diss.connect(dissLp); dissLp.connect(this.dissGain); this.dissGain.connect(this.master!);
   386→    this.droneOsc.start(); dr2.start(); diss.start();
   387→    // monster growl loop (gain driven per-frame)
   388→    this.growlOsc = ctx.createOscillator(); this.growlOsc.type = 'sawtooth'; this.growlOsc.frequency.value = 58;
   389→    const growlLp = ctx.createBiquadFilter(); growlLp.type = 'lowpass'; growlLp.frequency.value = 190;
   390→    const growlLfo = ctx.createOscillator(); growlLfo.frequency.value = 5.2;
   391→    const growlLfoG = ctx.createGain(); growlLfoG.gain.value = 14;

   425→  update(dt: number) {
   426→    if (!this.ctx || !this.droneGain || !this.droneOsc || !this.dissGain) return;
   427→    const t = this.ctx.currentTime;
   428→    this.droneGain.gain.setTargetAtTime(0.04 + this.danger * 0.09, t, 0.4);
   429→    this.droneOsc.frequency.setTargetAtTime(46 + this.danger * 10, t, 0.6);
   430→    this.dissGain.gain.setTargetAtTime(this.danger * 0.045, t, 0.5);
   431→    if (this.windGain) this.windGain.gain.setTargetAtTime(0.10 + this.danger * 0.08, t, 1.2);
   432→    // heartbeat
   433→    if (this.danger > 0.25) {
   434→      this.heartTimer -= dt;
   435→      if (this.heartTimer <= 0) {
   436→        const bpm = 42 + this.danger * 78;
   437→        this.heartTimer = 60 / bpm;
   438→        this.thump(0.5 + this.danger * 0.5);
   439→        setTimeout(() => this.thump(0.35 + this.danger * 0.35), 150);
   440→      }
   441→    }
   442→  }

   444→  private thump(vol: number) {
   445→    const ctx = this.ctx; if (!ctx || !this.master) return;
   446→    const o = ctx.createOscillator(); o.type = 'sine';
   447→    o.frequency.setValueAtTime(58, ctx.currentTime);

   452→    o.connect(g); g.connect(this.master);
   453→    o.start(); o.stop(ctx.currentTime + 0.25);
   454→  }

   456→  private env(node: AudioNode, vol: number, dur: number, attack = 0.005): GainNode {
   457→    const ctx = this.ctx!;

   466→  // spatial helper: returns [gain, pan]
   467→  spatial(listener: THREE.Vector3 | null, listenerYaw: number, pos: THREE.Vector3 | null, base: number): [number, number] {
   468→    if (!listener || !pos) return [base, 0];
   469→    const dx = pos.x - listener.x, dz = pos.z - listener.z;
   470→    const d = Math.hypot(dx, dz);
   471→    const gain = base * clamp(1 - d / 38, 0, 1) ** 1.4;
   472→    const ang = Math.atan2(dx, dz) - listenerYaw;
   473→    const pan = clamp(Math.sin(ang), -1, 1) * clamp(1 - d / 40, 0, 1);
   474→    return [gain, pan];
   475→  }

   781→/* ---------------- input ---------------- */
   782→export class Input {
   783→  keys: Record<string, boolean> = {};
   784→  buttons: Record<string, boolean> = {}; // unified action buttons (touch + kb)
   785→  look = { x: 0, y: 0 }; // consumed each frame
   786→  touchMove = { x: 0, y: 0 };
   787→  pointerLocked = false;
   788→  private el: HTMLElement | null = null;
   789→  onKeyExtra: ((code: string) => void) | null = null;
   790→
   791→  attach(el: HTMLElement) {
   792→    this.el = el;
   793→    window.addEventListener('keydown', this.kd);
   794→    window.addEventListener('keyup', this.ku);
   795→    el.addEventListener('mousedown', this.md);
   796→    window.addEventListener('mouseup', this.mu);
   797→    window.addEventListener('mousemove', this.mm);
   798→    document.addEventListener('pointerlockchange', this.plc);
   799→    el.addEventListener('contextmenu', (e) => e.preventDefault());
   800→  }
   801→  detach() {
   802→    window.removeEventListener('keydown', this.kd);
   803→    window.removeEventListener('keyup', this.ku);
   804→    window.removeEventListener('mouseup', this.mu);
   805→    window.removeEventListener('mousemove', this.mm);
   806→    document.removeEventListener('pointerlockchange', this.plc);
   807→  }
   808→  private kd = (e: KeyboardEvent) => {
   809→    if (e.repeat) { return; }
   810→    const code = normalizeKey(e);
   811→    this.keys[code] = true;
   812→    if (this.onKeyExtra) this.onKeyExtra(code);
   813→    if (['Space', 'Tab'].includes(code)) e.preventDefault();
   814→  };
   815→  private ku = (e: KeyboardEvent) => { this.keys[normalizeKey(e)] = false; };
   816→  private md = (e: MouseEvent) => {
   817→    if (!this.pointerLocked && this.el) { void this.el.requestPointerLock?.(); }
   818→    if (e.button === 0) this.buttons['attack'] = true;
   819→    if (e.button === 2) this.buttons['aim'] = true;
   820→  };
   821→  private mu = (e: MouseEvent) => {
   822→    if (e.button === 0) this.buttons['attack'] = false;
   823→    if (e.button === 2) this.buttons['aim'] = false;
   824→  };
   825→  private mm = (e: MouseEvent) => {
   826→    if (!this.pointerLocked) return;
   827→    this.look.x += e.movementX * 0.0022;
   828→    this.look.y += e.movementY * 0.0022;
   829→  };
   830→  private plc = () => { this.pointerLocked = document.pointerLockElement === this.el; };
   831→
   832→  requestLock() { if (this.el && !this.pointerLocked) void this.el.requestPointerLock?.(); }
   833→  exitLock() { if (document.pointerLockElement) document.exitPointerLock(); }
   834→
   835→  // touch hooks (called from ui.ts)
   836→  touchLook(dx: number, dy: number) { this.look.x += dx * 0.0045; this.look.y += dy * 0.0045; }
   837→  touchMoveVec(x: number, y: number) { this.touchMove.x = x; this.touchMove.y = y; }
   838→  setButton(name: string, down: boolean) { this.buttons[name] = down; if (down && this.onKeyExtra) this.onKeyExtra('T_' + name); }
   839→
   840→  consumeLook(): { x: number, y: number } {
   841→    const r = { x: this.look.x, y: this.look.y };
   842→    this.look.x = 0; this.look.y = 0;
   843→    return r;
   844→  }
   845→  moveVector(): { x: number, y: number } {
   846→    let x = 0, y = 0;
   847→    if (this.keys['KeyW'] || this.keys['ArrowUp']) y -= 1;
   848→    if (this.keys['KeyS'] || this.keys['ArrowDown']) y += 1;
   849→    if (this.keys['KeyA'] || this.keys['ArrowLeft']) x -= 1;
   850→    if (this.keys['KeyD'] || this.keys['ArrowRight']) x += 1;
   851→    const len = Math.hypot(x, y);
   852→    if (len > 1) { x /= len; y /= len; }
   853→    if (Math.hypot(this.touchMove.x, this.touchMove.y) > 0.12) { x = this.touchMove.x; y = this.touchMove.y; }
   854→    return { x, y };
   855→  }
   856→  down(code: string) { return !!this.keys[code] || !!this.buttons[code]; }
   857→}
