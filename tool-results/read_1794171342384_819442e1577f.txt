   134→   134→    @Override
   135→   135→    public void doFrame(long n) {
   136→   136→        float dt = lastN == 0 ? 0.016f : Math.min(0.05f, (n - lastN) / 1e9f);
   137→   137→        lastN = n;
   138→   138→        t += dt;
   139→   139→        screenT += dt;
   140→   140→        if (lightning > 0) lightning -= dt * 2.4f;
   141→   141→        lightningNext -= dt;
   142→   142→        if (lightningNext <= 0) {
   143→   143→            lightningNext = 9 + rnd.nextFloat() * 15;
   144→   144→            lightning = 0.7f + rnd.nextFloat() * 0.3f;
   145→   145→            audio.play(rnd.nextBoolean() ? "thunder" : "thunder_far", 0.55f, 0.1f);
   146→   146→        }
   147→   147→        if (toastT > 0) toastT -= dt;
   148→   148→        audio.update(dt);
   149→   149→        // menu music
   150→   150→        if (screen == SCR_START) audio.music("wind_loop", 0.5f);
   151→   151→        else audio.music("music_menu", 0.6f);
   152→   152→        invalidate();
   153→   153→        Choreographer.getInstance().postFrameCallback(this);
   154→   154→    }

   166→   166→    public void tap(float x, float y) {
   167→   167→        if (screen == SCR_START) {
   168→   168→            go(SCR_MAIN);
   169→   169→            audio.play("start_game");
   170→   170→            return;
   171→   171→        }
   172→   172→        if (screen == SCR_CREDITS) {
   173→   173→            go(SCR_MAIN);
   174→   174→            return;
   175→   175→        }
   176→   176→        if (screen == SCR_ACH) {
   177→   177→            go(SCR_MAIN);
   178→   178→            return;
   179→   179→        }
   180→   180→        if (screen == SCR_EXIT) {
   181→   181→            if (y > h * 0.6f) {
   182→   182→                if (x < w / 2) {
   183→   183→                    act.finishAffinity();
   184→   184→                } else {
   185→   185→                    go(SCR_MAIN);
   186→   186→                }
   187→   187→            }
   188→   188→            return;
   189→   189→        }
   190→   190→        // buttons
   191→   191→        for (int i = 0; i < buttons.size(); i++) {
   192→   192→            MBtn b = buttons.get(i);
   193→   193→            if (!b.enabled) continue;
   194→   194→            if (x >= b.x - b.bw / 2 && x <= b.x + b.bw / 2 && y >= b.y && y <= b.y + b.bh) {
   195→   195→                b.press = 1;
   196→   196→                audio.play("ui_select");
   197→   197→                if (b.run != null) b.run.run();
   198→   198→                return;
   199→   199→            }
   200→   200→        }
   201→   201→        // settings interactions
   202→   202→        if (screen == SCR_SETTINGS) settingsTap(x, y);
   203→   203→        if (screen == SCR_SAVES) savesTap(x, y);
   204→   204→        // back area (top-right corner)
   205→   205→        if (x > w - 70 * dp && y < 60 * dp) {
   206→   206→            go(SCR_MAIN);
   207→   207→            audio.play("ui_back");
   208→   208→        }
   209→   209→    }

   217→   217→    private void settingsTap(float x, float y) {
   218→   218→        float y0 = h * 0.30f;
   219→   219→        float sh = 40 * dp;
   220→   220→        for (int i = 0; i < 4; i++) {
   221→   221→            float sy = y0 + i * sh;
   222→   222→            if (y >= sy && y <= sy + 22 * dp && x >= 130 * dp && x <= w - 130 * dp) {
   223→   223→                float v = (x - 130 * dp) / (w - 260 * dp);
   224→   224→                v = Math.max(0.02f, Math.min(1f, v));
   225→   225→                switch (i) {
   226→   226→                    case 0: cfg.sens = 0.2f + v * 2.8f; break;
   227→   227→                    case 1: cfg.volMaster = v; break;
   228→   228→                    case 2: cfg.volMusic = v; break;
   229→   229→                    case 3: cfg.volSfx = v; break;
   230→   230→                }
   231→   231→                audio.play("ui_move", 0.4f);
   232→   232→                SaveIO.saveConfig(act, cfg);
   233→   233→                return;
   234→   234→            }
   235→   235→        }
   236→   236→        float qy = y0 + 4 * sh;
   237→   237→        for (int i = 0; i < 3; i++) {
   238→   238→            float bx = w * 0.5f + (i - 1) * 90 * dp;
   239→   239→            if (x >= bx - 40 * dp && x <= bx + 40 * dp && y >= qy && y <= qy + 26 * dp) {
   240→   240→                cfg.quality = i;
   241→   241→                SaveIO.saveConfig(act, cfg);
   242→   242→                audio.play("ui_select");
   243→   243→                return;
   244→   244→            }
   245→   245→        }
   246→   246→        for (int i = 0; i < 5; i++) {
   247→   247→            float ty = qy + 40 * dp + i * 32 * dp;
   248→   248→            if (x >= w - 150 * dp && x <= w - 40 * dp && y >= ty && y <= ty + 26 * dp) {
   249→   249→                switch (i) {
   250→   250→                    case 0: cfg.fog = !cfg.fog; break;
   251→   251→                    case 1: cfg.grain = !cfg.grain; break;
   252→   252→                    case 2: cfg.invertLook = !cfg.invertLook; break;
   253→   253→                    case 3: cfg.vibrate = !cfg.vibrate; break;
   254→   254→                    case 4: cfg.showFps = !cfg.showFps; break;
   255→   255→                }
   256→   256→                audio.play("ui_select", 0.7f);
   257→   257→                SaveIO.saveConfig(act, cfg);
   258→   258→                return;
   259→   259→            }
   260→   260→        }
   261→   261→    }

   263→   263→    private void savesTap(float x, float y) {
   264→   264→        float bw = Math.min(w * 0.62f, 460 * dp);
   265→   265→        float bh = 52 * dp;
   266→   266→        float x0 = w / 2 - bw / 2;
   267→   267→        float y0 = h * 0.24f;
   268→   268→        int idx = (int) ((y - y0) / (bh + 12 * dp));
   269→   269→        if (idx < 0 || idx > 3 || x < x0 || x > x0 + bw) return;
   270→   270→        if (!saveHas[idx]) {
   271→   271→            // start new game into this slot
   272→   272→            if (idx > 0) act.startGame(false, "slot" + idx + ".json");
   273→   273→            return;
   274→   274→        }
   275→   275→        // load zone vs delete zone
   276→   276→        if (x < x0 + bw - 60 * dp) {
   277→   277→            act.startGame(true, idx == 0 ? "slot_auto.json" : "slot" + idx + ".json");
   278→   278→        } else if (idx > 0) {
   279→   279→            SaveIO.delete(act, "slot" + idx + ".json");
   280→   280→            refreshSaves();
   281→   281→            audio.play("ui_back");
   282→   282→        }
   283→   283→    }

   352→   352→                new int[]{0x50A02020, 0x28801A1A, 0x00801A1A}, null, Shader.TileMode.CLAMP));
   353→   353→        c.drawCircle(mx, my, 90 * dp, p);
   354→   354→        p.setShader(null);
   355→   355→        p.setColor(0xFFB02828);
   356→   356→        c.drawCircle(mx, my, 40 * dp, p);

   378→   378→            float tx = w * (0.04f + i * 0.155f) + parX * (0.3f + i * 0.05f);
   379→   379→            float baseY = h * (0.82f + 0.02f * ((i * 7) % 3));
   380→   380→            float th = h * (0.13f + 0.03f * ((i * 5) % 4));

   382→   382→            c.drawLine(tx + 2 * dp, baseY - th * 0.5f, tx - 16 * dp, baseY - th * 0.75f, p);

   386→   386→        for (int i = 0; i < 3; i++) {
   387→   387→            float fy = h * (0.68f + i * 0.09f);

   390→   390→                    Color.argb(0, 130, 130, 145), Color.argb((int) (38 * (1 + lightning)), 130, 130, 148),
   391→   391→                    Shader.TileMode.CLAMP));
   392→   392→            p.setColor(Color.argb((int) (30 + 10 * i + 60 * lightning), 120 + (int) (40 * lightning),
   393→   393→                    120 + (int) (42 * lightning), 135 + (int) (48 * lightning)));
   394→   394→            Path fog = new Path();
   395→   395→            fog.moveTo(-20, h);
   396→   396→            float ph1 = fogSeed[i];
   397→   397→            for (float fx = -20; fx <= w + 40; fx += 24) {
   398→   398→                float yy = fy + (float) Math.sin(fx * 0.006f / dp * 0.3f + t * (0.25f + i * 0.12f) + ph1) * (14 + i * 8) * dp
   399→   399→                        + (float) Math.sin(fx * 0.0017f + t * 0.1f + ph1 * 2f) * 20 * dp;

   406→   406→        }
   407→   407→        // dust
   408→   408→        for (float[] d : dust) {
   409→   409→            d[0] += d[2] * 0.016f;
   410→   410→            if (d[0] < 0) d[0] += 1;
   411→   411→            if (d[0] > 1) d[0] -= 1;
   412→   412→            float yy = d[1] * h + 10 * dp * (float) Math.sin(t * 0.6f + d[3]);
   413→   413→            p.setColor(Color.argb((int) (30 + 22 * Math.sin(t * 2 + d[3])), 210, 205, 215));
   414→   414→            c.drawPoint(d[0] * w, yy, p);
   415→   415→        }
   416→   416→    }
