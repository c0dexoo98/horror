   260→  private screen(id: string, inner?: HTMLElement[]): HTMLElement {
   261→    const s = document.createElement('div');
   262→    s.id = 'hv-' + id;
   263→    s.className = 'hv-screen hv-hidden';
   264→    if (inner) inner.forEach(i => s.appendChild(i));
   265→    this.root.appendChild(s);
   266→    this.screens[id] = s;
   267→    return s;
   268→  }
   269→  show(name: string | null) {
   270→    for (const k of Object.keys(this.screens)) this.screens[k].classList.toggle('hv-hidden', k !== name);
   271→    if (name) this.input.exitLock();
   272→  }

   663→  private bindTouch(settings: Settings) {
   664→    this.applyTouchSetting(settings);
   665→    // joystick
   666→    let stickId = -1;
   667→    const stickCenter = { x: 0, y: 0 };
   668→    const R = 52;
   669→    const zoneL = document.createElement('div');
   670→    zoneL.className = 'zone';
   671→    zoneL.style.cssText = 'left:0;bottom:0;width:45%;height:70%;';
   672→    zoneL.addEventListener('touchstart', (e) => {
   673→      e.preventDefault();
   674→      const t = e.changedTouches[0];
   675→      if (stickId >= 0) return;
   676→      stickId = t.identifier;
   677→      stickCenter.x = t.clientX; stickCenter.y = t.clientY;
   678→      this.stickBase.classList.remove('hv-hidden');
   679→      this.stickNub.classList.remove('hv-hidden');
   680→      this.stickBase.style.left = (t.clientX - 60) + 'px';
   681→      this.stickBase.style.top = (t.clientY - 60) + 'px';
   682→      this.stickNub.style.left = (t.clientX - 26) + 'px';
   683→      this.stickNub.style.top = (t.clientY - 26) + 'px';
   684→    }, { passive: false });
   685→    zoneL.addEventListener('touchmove', (e) => {
   686→      e.preventDefault();
   687→      for (const t of Array.from(e.changedTouches)) {
   688→        if (t.identifier !== stickId) continue;
   689→        let dx = t.clientX - stickCenter.x, dy = t.clientY - stickCenter.y;
   690→        const d = Math.hypot(dx, dy);
   691→        if (d > R) { dx = dx / d * R; dy = dy / d * R; }
   692→        this.stickNub.style.left = (stickCenter.x + dx - 26) + 'px';
   693→        this.stickNub.style.top = (stickCenter.y + dy - 26) + 'px';
   694→        this.input.touchMoveVec(dx / R, dy / R);
   695→      }
   696→    }, { passive: false });
   697→    const stickEnd = (e: TouchEvent) => {
   698→      for (const t of Array.from(e.changedTouches)) {
   699→        if (t.identifier === stickId) {
   700→          stickId = -1;
   701→          this.input.touchMoveVec(0, 0);
   702→          this.stickBase.classList.add('hv-hidden');
   703→          this.stickNub.classList.add('hv-hidden');
   704→        }
   705→      }
   706→    };
   707→    zoneL.addEventListener('touchend', stickEnd);
   708→    zoneL.addEventListener('touchcancel', stickEnd);
   709→    this.touchWrap.appendChild(zoneL);
   710→    // look zone
   711→    const lookZone = document.createElement('div');
   712→    lookZone.id = 'hv-lookzone';
   713→    let lookId = -1;
   714→    const last = { x: 0, y: 0 };
   715→    lookZone.addEventListener('touchstart', (e) => {
   716→      e.preventDefault();
   717→      const t = e.changedTouches[0];
   718→      if (lookId >= 0) return;
   719→      lookId = t.identifier;
   720→      last.x = t.clientX; last.y = t.clientY;
   721→    }, { passive: false });
   722→    lookZone.addEventListener('touchmove', (e) => {
   723→      e.preventDefault();
   724→      for (const t of Array.from(e.changedTouches)) {
   725→        if (t.identifier !== lookId) continue;
   726→        this.input.touchLook(t.clientX - last.x, t.clientY - last.y);
   727→        last.x = t.clientX; last.y = t.clientY;
   728→      }
   729→    }, { passive: false });
   730→    const lookEnd = (e: TouchEvent) => {
   731→      for (const t of Array.from(e.changedTouches)) if (t.identifier === lookId) lookId = -1;
   732→    };
   733→    lookZone.addEventListener('touchend', lookEnd);
   734→    lookZone.addEventListener('touchcancel', lookEnd);
   735→    this.touchWrap.appendChild(lookZone);
   736→    // buttons
   737→    const mkTBtn = (label: string, style: string, onDown: () => void, onUp?: () => void, toggle?: () => boolean) => {
   738→      const b = document.createElement('button');
   739→      b.className = 'hv-tbtn';
   740→      b.textContent = label;
   741→      b.setAttribute('style', style);
   742→      b.addEventListener('touchstart', (e) => {
   743→        e.preventDefault();
   744→        b.classList.add('on');
   745→        onDown();

   791→function mkBtn(label: string, onClick: () => void, primary = false): HTMLButtonElement {
   792→  const b = document.createElement('button');
   793→  b.className = 'hv-btn' + (primary ? ' primary' : '');

   795→  b.addEventListener('click', onClick);
   796→  return b;
   797→}
   798→function mkRow(parent: HTMLElement, label: string, control: HTMLElement) {
   799→  const row = document.createElement('div');
   800→  row.className = 'hv-row';
   801→  const l = document.createElement('label');
   802→  l.textContent = label;
   803→  row.append(l, control);
   804→  parent.appendChild(row);
   805→}
   806→function mkSelect(opts: string[], val: string, onChange: (v: string) => void): HTMLSelectElement {
   807→  const s = document.createElement('select');
   808→  s.className = 'hv-select';
   809→  opts.forEach(o => {
   810→    const op = document.createElement('option');
   811→    op.value = o;
   812→    op.textContent = o.toUpperCase();
   813→    if (o === val) op.selected = true;
   814→    s.appendChild(op);
   815→  });
   816→  s.onchange = () => onChange(s.value);
   817→  return s;
   818→}
   819→function mkRange(min: number, max: number, val: number, onChange: (v: number) => void): HTMLInputElement {
   820→  const r = document.createElement('input');
   821→  r.type = 'range';
   822→  r.className = 'hv-range';
   823→  r.min = String(min); r.max = String(max);
   824→  r.step = '0.05';
   825→  r.value = String(val);
   826→  r.style.width = '140px';
   827→  r.oninput = () => onChange(parseFloat(r.value));
   828→  return r;
   829→}
   830→function mkCheck(val: boolean, onChange: (v: boolean) => void): HTMLInputElement {

   834→  c.style.accentColor = '#8f1d1d';
   835→  c.onchange = () => onChange(c.checked);
   836→  return c;
   837→}
