    48→function boxMesh(
    49→  w: number, h: number, d: number, mat: THREE.Material,
    50→  x: number, y: number, z: number, ry = 0,
    51→  parent?: THREE.Object3D
    52→): THREE.Mesh {
    53→  const geo = new THREE.BoxGeometry(w, h, d);
    54→  const m = new THREE.Mesh(geo, mat);
    55→  m.position.set(x, y, z);
    56→  m.rotation.y = ry;
    57→  if (parent) parent.add(m);
    58→  return m;
    59→}

    61→export function aabbFromCenter(cx: number, cz: number, w: number, d: number): AABB {
    62→  return { minX: cx - w / 2, maxX: cx + w / 2, minZ: cz - d / 2, maxZ: cz + d / 2 };
    63→}

    65→export function resolveCollisions(pos: THREE.Vector3, radius: number, colliders: AABB[]) {
    66→  for (const c of colliders) {
    67→    const cx = Math.max(c.minX, Math.min(pos.x, c.maxX));
    68→    const cz = Math.max(c.minZ, Math.min(pos.z, c.maxZ));
    69→    const dx = pos.x - cx, dz = pos.z - cz;
    70→    const d2 = dx * dx + dz * dz;
    71→    if (d2 < radius * radius) {
    72→      if (d2 > 1e-8) {
    73→        const d = Math.sqrt(d2);
    74→        pos.x = cx + (dx / d) * radius;
    75→        pos.z = cz + (dz / d) * radius;
    76→      } else {
    77→        // inside: push out along smallest penetration axis
    78→        const px = Math.min(pos.x - c.minX, c.maxX - pos.x);
    79→        const pz = Math.min(pos.z - c.minZ, c.maxZ - pos.z);
    80→        if (px < pz) pos.x = (pos.x - c.minX < c.maxX - pos.x) ? c.minX - radius : c.maxX + radius;
    81→        else pos.z = (pos.z - c.minZ < c.maxZ - pos.z) ? c.minZ - radius : c.maxZ + radius;
    82→      }
    83→    }
    84→  }
    85→}

    87→const _ray = new THREE.Raycaster();
    88→export function raycastClear(from: THREE.Vector3, to: THREE.Vector3, occluders: THREE.Object3D[]): boolean {
    89→  const dir = new THREE.Vector3().subVectors(to, from);
    90→  const len = dir.length();
    91→  if (len < 0.01) return true;
    92→  dir.normalize();
    93→  _ray.set(from, dir);
    94→  _ray.far = len - 0.1;
    95→  const hits = _ray.intersectObjects(occluders, false);
    96→  return hits.length === 0;
    97→}

   307→  // house with a door gap on one side. N = -z, S = +z, W = -x, E = +x
   308→  interface HouseCfg {
   309→    x: number; z: number; w: number; d: number; h: number;
   310→    door: 'N' | 'S' | 'E' | 'W';
   311→    doorW?: number;
   312→    locked?: string; // door id if locked
   313→    windows?: { side: 'N' | 'S' | 'E' | 'W'; off: number; lit?: boolean }[];
   314→    interior?: [string, number, number, number?][];
   315→    roof?: boolean;
   316→    stone?: boolean;
   317→  }
   318→  const makeHouse = (cfg: HouseCfg) => {
   319→    const { x, z, w, d, h } = cfg;
   320→    const mat = cfg.stone ? matStone : matPlank;
   321→    const dw = cfg.doorW ?? 1.4;
   322→    const wallSegs = (side: 'N' | 'S' | 'E' | 'W', winOffs: number[]) => {
   323→      const horiz = side === 'N' || side === 'S';
   324→      const fixed = side === 'N' ? z - d / 2 : side === 'S' ? z + d / 2 : side === 'W' ? x - w / 2 : x + w / 2;
   325→      const len = horiz ? w : d;
   326→      const isDoorSide = side === cfg.door;
   327→      const segments: [number, number][] = isDoorSide
   328→        ? [[-len / 2, -dw / 2], [dw / 2, len / 2]]
   329→        : [[-len / 2, len / 2]];
   330→      for (const [a, b] of segments) {
   331→        const segLen = b - a;
   332→        if (segLen < 0.05) continue;
   333→        const mid = (a + b) / 2;
   334→        if (horiz) wall(x + mid, h / 2, fixed, segLen, h, 0.3, mat);
   335→        else wall(fixed, h / 2, z + mid, 0.3, h, segLen, mat);
   336→      }
   337→      // lintel above door
   338→      if (isDoorSide) {
   339→        if (horiz) wall(x, h - 0.35, fixed, dw, 0.7, 0.3, mat, false);
   340→        else wall(fixed, h - 0.35, z, 0.3, 0.7, dw, mat, false);
   341→      }
   342→      for (const off of winOffs) {
   343→        if (isDoorSide && Math.abs(off) < 2) continue;
   344→        const wy = h * 0.55;
   345→        if (horiz) windowPane(x + off, wy, fixed + (side === 'N' ? -0.18 : 0.18), 0, !!cfg.windows?.find(ww => ww.off === off)?.lit);
   346→        else windowPane(fixed + (side === 'W' ? -0.18 : 0.18), wy, z + off, Math.PI / 2, !!cfg.windows?.find(ww => ww.off === off)?.lit);
   347→      }
   348→    };
   349→    (['N', 'S', 'E', 'W'] as const).forEach(side => {
   350→      const wins = (cfg.windows || []).filter(v => v.side === side).map(v => v.off);
   351→      wallSegs(side, wins);
   352→    });
   353→    // floor
   354→    const floor = new THREE.Mesh(new THREE.PlaneGeometry(w, d), matWood);
   355→    floor.rotation.x = -Math.PI / 2;
   356→    floor.position.set(x, 0.03, z);
   357→    root.add(floor);
   358→    if (cfg.roof !== false) gableRoof(x, h, z, w, d);
   359→    (cfg.interior || []).forEach(([kind, ix, iz, r]) => furn(kind, x + ix, z + iz, r || 0));
   360→  };

   362→  // locked door leaf
   363→  const makeDoor = (id: string, x: number, z: number, ry: number, locked: boolean) => {
   364→    const pivot = new THREE.Object3D();
   365→    pivot.position.set(x, 0, z);
   366→    pivot.rotation.y = ry;
   367→    const leaf = boxMesh(1.36, 2.2, 0.12, matPlankDark, 0.68, 1.1, 0);
   368→    leaf.castShadow = shadows;
   369→    pivot.add(leaf);
   370→    const knob = boxMesh(0.1, 0.1, 0.16, matMetal, 1.15, 1.05, 0);
   371→    pivot.add(knob);
   372→    root.add(pivot);
   373→    // collider approximated as closed door across gap
   374→    const horiz = Math.abs(Math.sin(ry)) < 0.5;
   375→    const col = horiz ? aabbFromCenter(x + 0.68 * Math.cos(ry), z, 1.4, 0.5) : aabbFromCenter(x, z + 0.68 * Math.cos(ry), 0.5, 1.4);
   376→    addCollider(col);
   377→    const d: DoorRef = { id, locked, open: false, pivot, collider: col };
   378→    doors.set(id, d);
   379→    interactables.push({
   380→      id: 'door_' + id,
   381→      pos: new THREE.Vector3(x, 1.2, z),
   382→      radius: 2.6,
   383→      prompt: locked ? 'Locked' : 'Open door',
   384→      enabled: true,
   385→      onInteract: () => { /* handled by quests */ },
   386→    });
   387→    return d;
   388→  };
