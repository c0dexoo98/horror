    80→    public MenuView(Context c, MenuActivity act, GameConfig cfg, AudioSys audio) {
    81→        super(c);
    82→        this.act = act;
    83→        this.cfg = cfg;
    84→        this.audio = audio;
    85→        dp = c.getResources().getDisplayMetrics().density;
    86→        for (float[] s : stars) {
    87→            s[0] = rnd.nextFloat();
    88→            s[1] = rnd.nextFloat() * 0.55f;
    89→            s[2] = rnd.nextFloat() * 6.28f;
    90→        }
    91→        for (float[] d : dust) {
    92→            d[0] = rnd.nextFloat();
    93→            d[1] = rnd.nextFloat();
    94→            d[2] = (rnd.nextFloat() - 0.5f) * 0.02f;
    95→            d[3] = rnd.nextFloat() * 6.28f;
    96→        }
    97→        for (int i = 0; i < 3; i++) fogSeed[i] = rnd.nextFloat() * 100;
    98→        try {
    99→            sm = (SensorManager) c.getSystemService(Context.SENSOR_SERVICE);
   100→            sensorListener = new SensorEventListener() {
   101→                @Override
   102→                public void onSensorChanged(SensorEvent e) {
   103→                    if (e.sensor.getType() == Sensor.TYPE_ACCELEROMETER) {
   104→                        parX += ((-e.values[0] / 9f) * 14 * dp - parX) * 0.08f;
   105→                        parY += ((e.values[1] / 9f) * 8 * dp - parY) * 0.08f;
   106→                    }
   107→                }
   108→
   109→                @Override
   110→                public void onAccuracyChanged(Sensor s, int a) {
   111→                }
   112→            };
   113→            sm.registerListener(sensorListener, sm.getDefaultSensor(Sensor.TYPE_ACCELEROMETER), SensorManager.SENSOR_DELAY_GAME);
   114→        } catch (Throwable ignored) {
   115→            sm = null;
   116→        }
   117→        Choreographer.getInstance().postFrameCallback(this);
   118→        refreshSaves();
   119→    }

   134→    @Override
   135→    public void doFrame(long n) {
   136→        float dt = lastN == 0 ? 0.016f : Math.min(0.05f, (n - lastN) / 1e9f);
   137→        lastN = n;
   138→        t += dt;
   139→        screenT += dt;
   140→        if (lightning > 0) lightning -= dt * 2.4f;
   141→        lightningNext -= dt;
   142→        if (lightningNext <= 0) {
   143→            lightningNext = 9 + rnd.nextFloat() * 15;
   144→            lightning = 0.7f + rnd.nextFloat() * 0.3f;
   145→            audio.play(rnd.nextBoolean() ? "thunder" : "thunder_far", 0.55f, 0.1f);
   146→        }
   147→        if (toastT > 0) toastT -= dt;
   148→        audio.update(dt);
   149→        // menu music
   150→        if (screen == SCR_START) audio.music("wind_loop", 0.5f);
   151→        else audio.music("music_menu", 0.6f);
   152→        invalidate();
   153→        Choreographer.getInstance().postFrameCallback(this);
   154→    }

   166→    public void tap(float x, float y) {
   167→        if (screen == SCR_START) {
   168→            go(SCR_MAIN);
   169→            audio.play("start_game");
   170→            return;
   171→        }
   172→        if (screen == SCR_CREDITS) {
   173→            go(SCR_MAIN);
   174→            return;
   175→        }
   176→        if (screen == SCR_ACH) {
   177→            go(SCR_MAIN);
   178→            return;
   179→        }
   180→        if (screen == SCR_EXIT) {
   181→            if (y > h * 0.6f) {
   182→                if (x < w / 2) {
   183→                    act.finishAffinity();
   184→                } else {
   185→                    go(SCR_MAIN);
   186→                }
   187→            }
   188→            return;
   189→        }
   190→        // buttons
   191→        for (int i = 0; i < buttons.size(); i++) {
   192→            MBtn b = buttons.get(i);
   193→            if (!b.enabled) continue;
   194→            if (x >= b.x - b.bw / 2 && x <= b.x + b.bw / 2 && y >= b.y && y <= b.y + b.bh) {
   195→                b.press = 1;
   196→                audio.play("ui_select");
   197→                if (b.run != null) b.run.run();
   198→                return;
   199→            }
   200→        }
   201→        // settings interactions
   202→        if (screen == SCR_SETTINGS) settingsTap(x, y);
   203→        if (screen == SCR_SAVES) savesTap(x, y);
   204→        // back area (top-right corner)
   205→        if (x > w - 70 * dp && y < 60 * dp) {
   206→            go(SCR_MAIN);
   207→            audio.play("ui_back");
   208→        }
   209→    }

   217→    private void settingsTap(float x, float y) {
   218→        float y0 = h * 0.30f;
   219→        float sh = 40 * dp;
   220→        for (int i = 0; i < 4; i++) {
   221→            float sy = y0 + i * sh;
   222→            if (y >= sy && y <= sy + 22 * dp && x >= 130 * dp && x <= w - 130 * dp) {
   223→                float v = (x - 130 * dp) / (w - 260 * dp);
   224→                v = Math.max(0.02f, Math.min(1f, v));
   225→                switch (i) {
   226→                    case 0: cfg.sens = 0.2f + v * 2.8f; break;
   227→                    case 1: cfg.volMaster = v; break;
   228→                    case 2: cfg.volMusic = v; break;
   229→                    case 3: cfg.volSfx = v; break;
   230→                }
   231→                audio.play("ui_move", 0.4f);
   232→                SaveIO.saveConfig(act, cfg);
   233→                return;
   234→            }
   235→        }
   236→        float qy = y0 + 4 * sh;
   237→        for (int i = 0; i < 3; i++) {
   238→            float bx = w * 0.5f + (i - 1) * 90 * dp;
   239→            if (x >= bx - 40 * dp && x <= bx + 40 * dp && y >= qy && y <= qy + 26 * dp) {
   240→                cfg.quality = i;
   241→                SaveIO.saveConfig(act, cfg);
   242→                audio.play("ui_select");
   243→                return;
   244→            }
   245→        }
   246→        for (int i = 0; i < 5; i++) {
   247→            float ty = qy + 40 * dp + i * 32 * dp;
   248→            if (x >= w - 150 * dp && x <= w - 40 * dp && y >= ty && y <= ty + 26 * dp) {
   249→                switch (i) {
   250→                    case 0: cfg.fog = !cfg.fog; break;
   251→                    case 1: cfg.grain = !cfg.grain; break;
   252→                    case 2: cfg.invertLook = !cfg.invertLook; break;
   253→                    case 3: cfg.vibrate = !cfg.vibrate; break;
   254→                    case 4: cfg.showFps = !cfg.showFps; break;
   255→                }
   256→                audio.play("ui_select", 0.7f);
   257→                SaveIO.saveConfig(act, cfg);
   258→                return;
   259→            }
   260→        }
   261→    }

   263→    private void savesTap(float x, float y) {
   264→        float bw = Math.min(w * 0.62f, 460 * dp);
   265→        float bh = 52 * dp;
   266→        float x0 = w / 2 - bw / 2;
   267→        float y0 = h * 0.24f;
   268→        int idx = (int) ((y - y0) / (bh + 12 * dp));
   269→        if (idx < 0 || idx > 3 || x < x0 || x > x0 + bw) return;
   270→        if (!saveHas[idx]) {
   271→            // start new game into this slot
   272→            if (idx > 0) act.startGame(false, "slot" + idx + ".json");
   273→            return;
   274→        }
   275→        // load zone vs delete zone
   276→        if (x < x0 + bw - 60 * dp) {
   277→            act.startGame(true, idx == 0 ? "slot_auto.json" : "slot" + idx + ".json");
   278→        } else if (idx > 0) {
   279→            SaveIO.delete(act, "slot" + idx + ".json");
   280→            refreshSaves();
   281→            audio.play("ui_back");
   282→        }
   283→    }

   331→    private void drawBackground(Canvas c) {
   332→        if (videoBg) {
   333→            drawVideoBackground(c);
   334→            return;
   335→        }
   336→        // sky
   337→        p.setShader(new LinearGradient(0, 0, 0, h,
   338→                Color.rgb(9, 8, 14), Color.rgb(16, 13, 24), Shader.TileMode.CLAMP));
   339→        c.drawRect(0, 0, w, h, p);
   340→        p.setShader(null);
   341→        // stars
   342→        for (float[] s : stars) {
   343→            float tw = 0.5f + 0.5f * (float) Math.sin(t * 1.7 + s[2]);
   344→            if (tw > 0.4f) {
   345→                p.setColor(Color.argb((int) (140 * tw), 200, 200, 215));
   346→                c.drawPoint(s[0] * w + parX * 0.2f, s[1] * h + parY * 0.2f, p);
   347→            }
   348→        }
   349→        // moon
   350→        float mx = w * 0.78f + parX * 0.5f, my = h * 0.20f + parY * 0.5f;
   351→        p.setShader(new RadialGradient(mx, my, 90 * dp,
   352→                new int[]{0x50A02020, 0x28801A1A, 0x00801A1A}, null, Shader.TileMode.CLAMP));
   353→        c.drawCircle(mx, my, 90 * dp, p);
   354→        p.setShader(null);
   355→        p.setColor(0xFFB02828);
   356→        c.drawCircle(mx, my, 40 * dp, p);
   357→        p.setColor(0xFF982020);
   358→        c.drawCircle(mx - 10 * dp, my + 8 * dp, 12 * dp, p);
   359→        c.drawCircle(mx + 14 * dp, my - 10 * dp, 8 * dp, p);
   360→        if (lightning > 0) {
   361→            p.setColor(Color.argb((int) (70 * lightning), 120, 130, 170));
   362→            c.drawRect(0, 0, w, h, p);
   363→        }
   364→        // hills
   365→        p.setColor(0xFF0D0B12);
   366→        Path hill = new Path();
   367→        hill.moveTo(-50, h);
   368→        hill.quadTo(w * 0.2f, h * 0.72f, w * 0.5f, h * 0.80f);
   369→        hill.quadTo(w * 0.8f, h * 0.88f, w + 50, h * 0.74f);
   370→        hill.lineTo(w + 50, h);
   371→        hill.close();
   372→        c.drawPath(hill, p);
   373→        // mansion silhouette
   374→        drawMansion(c, w * 0.5f + parX, h * 0.66f + parY * 0.4f, 1f, 0xFF0A0810, lightning);
   375→        // trees
   376→        p.setColor(0xFF070610);
   377→        for (int i = 0; i < 7; i++) {
   378→            float tx = w * (0.04f + i * 0.155f) + parX * (0.3f + i * 0.05f);
   379→            float baseY = h * (0.82f + 0.02f * ((i * 7) % 3));
   380→            float th = h * (0.13f + 0.03f * ((i * 5) % 4));
   381→            c.drawLine(tx, baseY, tx + 6 * dp * ((i % 2) == 0 ? 1 : -1), baseY - th, p);
   382→            c.drawLine(tx + 2 * dp, baseY - th * 0.5f, tx - 16 * dp, baseY - th * 0.75f, p);
   383→            c.drawLine(tx + 2 * dp, baseY - th * 0.6f, tx + 20 * dp, baseY - th * 0.9f, p);
   384→        }
   385→        // fog layers
   386→        for (int i = 0; i < 3; i++) {
   387→            float fy = h * (0.68f + i * 0.09f);
   388→            float amp = 0.16f + 0.07f * i;
   389→            p.setShader(new LinearGradient(0, fy - 40 * dp, 0, fy + 60 * dp,
   390→                    Color.argb(0, 130, 130, 145), Color.argb((int) (38 * (1 + lightning)), 130, 130, 148),
   391→                    Shader.TileMode.CLAMP));
   392→            p.setColor(Color.argb((int) (30 + 10 * i + 60 * lightning), 120 + (int) (40 * lightning),
   393→                    120 + (int) (42 * lightning), 135 + (int) (48 * lightning)));
   394→            Path fog = new Path();
   395→            fog.moveTo(-20, h);
   396→            float ph1 = fogSeed[i];
   397→            for (float fx = -20; fx <= w + 40; fx += 24) {
   398→                float yy = fy + (float) Math.sin(fx * 0.006f / dp * 0.3f + t * (0.25f + i * 0.12f) + ph1) * (14 + i * 8) * dp
   399→                        + (float) Math.sin(fx * 0.0017f + t * 0.1f + ph1 * 2f) * 20 * dp;

   402→            fog.lineTo(w + 40, h);
   403→            fog.close();

   408→        for (float[] d : dust) {
   409→            d[0] += d[2] * 0.016f;
   410→            if (d[0] < 0) d[0] += 1;
   411→            if (d[0] > 1) d[0] -= 1;
   412→            float yy = d[1] * h + 10 * dp * (float) Math.sin(t * 0.6f + d[3]);
   413→            p.setColor(Color.argb((int) (30 + 22 * Math.sin(t * 2 + d[3])), 210, 205, 215));
   414→            c.drawPoint(d[0] * w, yy, p);
   415→        }
   416→    }

   418→    /**
   419→     * Atmosphere painted OVER the looping background video: soft scrims so the
   420→     * UI stays readable, plus lightning flashes, drifting fog and dust that
   421→     * blend the footage into the game's horror mood.
   422→     */
   423→    private void drawVideoBackground(Canvas c) {
   424→        // vertical scrim: darker top & bottom bands, clearer window in the middle
   425→        p.setShader(new LinearGradient(0, 0, 0, h, new int[]{
   426→                Color.argb(150, 4, 4, 8),
   427→                Color.argb(38, 4, 4, 8),
   428→                Color.argb(20, 4, 4, 8),
   429→                Color.argb(120, 3, 3, 7)},
   430→                new float[]{0f, 0.32f, 0.62f, 1f}, Shader.TileMode.CLAMP));
   431→        c.drawRect(0, 0, w, h, p);
   432→        p.setShader(null);
   433→        // pulsing red vignette tint (heartbeat of the vale)
   434→        float beat = 0.5f + 0.5f * (float) Math.sin(t * 1.05);
   435→        p.setShader(new RadialGradient(w * 0.5f, h * 0.46f, Math.max(w, h) * 0.75f,
   436→                new int[]{0x00000000,
   437→                        Color.argb((int) (26 + 14 * beat), 20, 0, 2),
   438→                        Color.argb((int) (95 + 40 * beat), 8, 0, 1)},
   439→                null, Shader.TileMode.CLAMP));
   440→        c.drawRect(0, 0, w, h, p);
   441→        p.setShader(null);
   442→        // lightning flash over footage
   443→        if (lightning > 0) {
   444→            p.setColor(Color.argb((int) (52 * lightning), 120, 130, 170));
   445→            c.drawRect(0, 0, w, h, p);
   446→        }
   447→        // drifting fog (thinner so the footage stays visible)
   448→        for (int i = 0; i < 2; i++) {
   449→            float fy = h * (0.74f + i * 0.14f);
   450→            p.setColor(Color.argb((int) (20 + 8 * i + 40 * lightning), 120, 120, 135));
   451→            Path fog = new Path();
   452→            fog.moveTo(-20, h);
   453→            float ph1 = fogSeed[i];
   454→            for (float fx = -20; fx <= w + 40; fx += 26) {
   455→                float yy = fy + (float) Math.sin(fx * 0.006f / dp * 0.3f + t * (0.22f + i * 0.1f) + ph1) * (10 + i * 7) * dp
   456→                        + (float) Math.sin(fx * 0.0017f + t * 0.1f + ph1 * 2f) * 16 * dp;
   457→                fog.lineTo(fx, yy);
   458→            }
   459→            fog.lineTo(w + 40, h);
   460→            fog.close();
   461→            c.drawPath(fog, p);
   462→        }
   463→        // floating dust catching the light
   464→        for (float[] d : dust) {
   465→            d[0] += d[2] * 0.016f;
   466→            if (d[0] < 0) d[0] += 1;
   467→            if (d[0] > 1) d[0] -= 1;
   468→            float yy = d[1] * h + 10 * dp * (float) Math.sin(t * 0.6f + d[3]);
   469→            p.setColor(Color.argb((int) (26 + 20 * Math.sin(t * 2 + d[3])), 210, 205, 215));
   470→            c.drawPoint(d[0] * w, yy, p);
   471→        }
   472→    }
