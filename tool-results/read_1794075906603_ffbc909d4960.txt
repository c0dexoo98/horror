   118→   118→  /* ---------- pickups ---------- */
   119→   119→  private buildPickups() {
   120→   120→    const paperMat = new THREE.MeshStandardMaterial({ map: getTexture('paper'), roughness: 0.9 });
   121→   121→    for (const p of PICKUPS) {
   122→   122→      const g = new THREE.Group();
   123→   123→      if (NOTE_IDS.has(p.type)) {
   124→   124→        const m = new THREE.Mesh(new THREE.BoxGeometry(0.3, 0.02, 0.4), paperMat);
   125→   125→        g.add(m);
   126→   126→      } else if (p.type === 'church_key' || p.type === 'iron_key') {
   127→   127→        const mat = new THREE.MeshStandardMaterial({ color: PICKUP_COLORS[p.type], metalness: 0.6, roughness: 0.4 });
   128→   128→        const shank = new THREE.Mesh(new THREE.BoxGeometry(0.04, 0.04, 0.26), mat);
   129→   129→        shank.position.z = 0.06;
   130→   130→        const ring = new THREE.Mesh(new THREE.TorusGeometry(0.07, 0.02, 6, 12), mat);
   131→   131→        ring.position.z = -0.1;
   132→   132→        const teeth = new THREE.Mesh(new THREE.BoxGeometry(0.03, 0.09, 0.05), mat);
   133→   133→        teeth.position.set(0, -0.05, 0.16);
   134→   134→        g.add(shank, ring, teeth);
   135→   135→      } else if (p.type === 'amulet') {
   136→   136→        const mat = new THREE.MeshStandardMaterial({ color: PICKUP_COLORS[p.type], metalness: 0.7, roughness: 0.3, emissive: 0x3a1a6a, emissiveIntensity: 0.6 });
   137→   137→        g.add(new THREE.Mesh(new THREE.TorusGeometry(0.11, 0.03, 8, 16), mat));
   138→   138→        const gem = new THREE.Mesh(new THREE.SphereGeometry(0.05, 8, 8), new THREE.MeshStandardMaterial({ color: 0x1a0a2a, emissive: 0x6a2aff, emissiveIntensity: 1 }));
   139→   139→        gem.position.z = 0.09;
   140→   140→        g.add(gem);
   141→   141→      } else if (p.type === 'herb') {
   142→   142→        const mat = new THREE.MeshStandardMaterial({ color: PICKUP_COLORS[p.type], roughness: 1 });
   143→   143→        for (let i = 0; i < 3; i++) {
   144→   144→          const leaf = new THREE.Mesh(new THREE.ConeGeometry(0.03, 0.18, 5), mat);
   145→   145→          leaf.position.set((this.rnd() - 0.5) * 0.12, 0.09, (this.rnd() - 0.5) * 0.12);
   146→   146→          leaf.rotation.z = (this.rnd() - 0.5) * 0.8;
   147→   147→          g.add(leaf);
   148→   148→        }
   149→   149→      } else {
   150→   150→        const mat = new THREE.MeshStandardMaterial({ color: PICKUP_COLORS[p.type] || 0xaaaaaa, roughness: 0.6, metalness: p.type.startsWith('ammo') ? 0.4 : 0.1 });
   151→   151→        const m = new THREE.Mesh(new THREE.BoxGeometry(0.22, 0.16, 0.16), mat);
   152→   152→        g.add(m);
   153→   153→        if (p.type.startsWith('ammo')) {
   154→   154→          for (let i = 0; i < 3; i++) {
   155→   155→            const bullet = new THREE.Mesh(new THREE.CylinderGeometry(0.018, 0.018, 0.09, 6), new THREE.MeshStandardMaterial({ color: 0xc9a24a, metalness: 0.7, roughness: 0.35 }));
   156→   156→            bullet.position.set(-0.06 + i * 0.06, 0.12, 0);
   157→   157→            g.add(bullet);
   158→   158→          }
   159→   159→        }
   160→   160→      }
   161→   161→      g.position.copy(p.pos);
   162→   162→      g.visible = !START_HIDDEN.has(p.id);
   163→   163→      this.world.root.add(g);
   164→   164→      this.pickupMeshes.set(p.id, g);
   165→   165→      const entry: Interactable = {
   166→   166→        id: p.id, pos: p.pos.clone(), radius: 2.1, prompt: NOTE_IDS.has(p.type) ? p.label : 'Take ' + p.label,
   167→   167→        enabled: !START_HIDDEN.has(p.id), once: true, highlight: g,
   168→   168→        onInteract: () => this.takePickup(p.id),
   169→   169→      };
   170→   170→      this.world.interactables.push(entry);
   171→   171→      if (START_HIDDEN.has(p.id)) entry.used = true;
   172→   172→    }
   173→   173→  }

   175→   175→  private takePickup(id: string) {
   176→   176→    const def = PICKUPS.find(p => p.id === id);
   177→   177→    if (!def || this.state.picked.includes(id)) return;
   178→   178→    this.state.picked.push(id);
   179→   179→    const mesh = this.pickupMeshes.get(id);
   180→   180→    if (mesh) mesh.visible = false;
   181→   181→    this.disableInteractable(id);
   182→   182→    if (NOTE_IDS.has(def.type)) {
   183→   183→      const note = NOTES.find(n => n.id === def.type);
   184→   184→      if (note) {
   185→   185→        if (!this.state.notesRead.includes(def.type)) this.state.notesRead.push(def.type);
   186→   186→        this.hooks.onNoteRead(note);
   187→   187→        this.checkWhispers();
   188→   188→        this.checkPriestNote(def.type);
   189→   189→      }
   190→   190→      this.hooks.onPickup(id, def.type);
   191→   191→      return;
   192→   192→    }
   193→   193→    this.hooks.onPickup(id, def.type);
   194→   194→    switch (def.type) {
   195→   195→      case 'church_key':
   196→   196→        this.hooks.message('Church Key acquired', true);
   197→   197→        this.setStage(1);
   198→   198→        break;
   199→   199→      case 'iron_key':
   200→   200→        this.hooks.message('Iron Key acquired', true);
   201→   201→        this.setStage(5);
   202→   202→        break;
   203→   203→      case 'amulet':
   204→   204→        this.hooks.message('The Drowned Amulet', true);
   205→   205→        this.setStage(4);
   206→   206→        break;
   207→   207→      case 'rifle_stock': this.hooks.message('Rifle Stock acquired'); break;
   208→   208→      case 'rifle_barrel': this.hooks.message('Rifle Barrel acquired'); break;
   209→   209→      case 'herb': {
   210→   210→        const count = ['pk_herb_1', 'pk_herb_2', 'pk_herb_3'].filter(h => this.state.picked.includes(h)).length;
   211→   211→        this.hooks.message(`Nightshade (${count}/3)`);
   212→   212→        break;
   213→   213→      }
   214→   214→      default: break;
   215→   215→    }
   216→   216→  }

   218→   218→  private checkPriestNote(noteId: string) {
   219→   219→    if (noteId === 'note_priest' && this.state.stage === 1) {
   220→   220→      this.setStage(2);
   221→   221→      this.hooks.subtitle('Three flames. Three calls. The braziers wait in the houses.', 6);
   222→   222→    }
   223→   223→  }
   224→   224→
   225→   225→  private checkWhispers() {
   226→   226→    if (this.state.whisperReward) return;
   227→   227→    if (this.state.notesRead.length >= NOTES.length) {
   228→   228→      this.state.whisperReward = true;
   229→   229→      this.hooks.message('All whispers collected: Revolver Ammo +6, Pills +2', true);
   230→   230→      this.hooks.subtitle('The last page reveals the gate-ward sigil. The village finally rests.', 6);
   231→   231→      this.hooks.netEmit('give', { items: { ammo_pistol: 6, pills: 2 } });
   232→   232→    }
   233→   233→  }

   235→   235→  /* ---------- world interactions ---------- */
   236→   236→  private wireWorld() {
   237→   237→    const byId = new Map<string, (typeof this.world.interactables)[number]>();
   238→   238→    for (const it of this.world.interactables) byId.set(it.id, it);
   239→   239→
   240→   240→    // doors
   241→   241→    for (const [id, door] of this.world.doors) {
   242→   242→      const it = byId.get('door_' + id);
   243→   243→      if (it) {
   244→   244→        it.prompt = door.locked ? 'Locked — needs a key' : 'Open door';
   245→   245→        it.onInteract = () => this.useDoor(id);
   246→   246→      }
   247→   247→    }
   248→   248→
   249→   249→    // braziers
   250→   250→    for (const b of this.world.braziers) {
   251→   251→      const it = byId.get(b.id);
   252→   252→      if (it) it.onInteract = () => this.lightBrazier(b.id);
   253→   253→    }
   254→   254→
   255→   255→    // well
   256→   256→    const well = byId.get('well_crank');
   257→   257→    if (well) well.onInteract = () => this.useWell();
   258→   258→
   259→   259→    // cemetery altar
   260→   260→    const altar = byId.get('cemetery_altar');
   261→   261→    if (altar) altar.onInteract = () => this.useAltar();
   262→   262→
   263→   263→    // gate
   264→   264→    const gate = byId.get('forest_gate');
   265→   265→    if (gate) gate.onInteract = () => this.useGate();
   266→   266→
   267→   267→    // stations: workbench + mortar
   268→   268→    this.world.interactables.push({
   269→   269→      id: 'workbench', pos: new THREE.Vector3(-4, 0.9, 28.4), radius: 2.6, prompt: 'Use workbench', enabled: true,
   270→   270→      onInteract: () => this.useWorkbench(),
   271→   271→    });
   272→   272→    this.world.interactables.push({
   273→   273→      id: 'mortar', pos: new THREE.Vector3(-25.2, 0.9, 8), radius: 2.6, prompt: 'Use mortar', enabled: true,
   274→   274→      onInteract: () => this.useMortar(),
   275→   275→    });
   276→   276→  }
   277→   277→
   278→   278→  useDoor(id: string) {
   279→   279→    const door = this.world.doors.get(id);
   280→   280→    if (!door) return;
   281→   281→    if (door.locked) {
   282→   282→      if (id === 'church') {
   283→   283→        if ((this.localPlayerRef?.inv.church_key || 0) > 0) {
   284→   284→          door.locked = false;
   285→   285→          this.hooks.message('Unlocked the church', true);
   286→   286→          this.openDoor(id);
   287→   287→        } else {
   288→   288→          this.hooks.subtitle('Locked. The keyhole is cold iron. The key is somewhere in the barn.', 4);
   289→   289→        }
   290→   290→      }
   291→   291→      return;
   292→   292→    }
   293→   293→    if (door.open) { this.closeDoor(id); } else { this.openDoor(id); }
   294→   294→  }
   295→   295→
   296→   296→  localPlayerRef: Player | null = null;
   297→   297→
   298→   298→  openDoor(id: string) {
   299→   299→    const door = this.world.doors.get(id);
   300→   300→    if (!door || door.open) return;
   301→   301→    door.open = true;
   302→   302→    door.pivot.rotation.y = -Math.PI * 0.55;
   303→   303→    const ci = this.world.colliders.indexOf(door.collider);
   304→   304→    if (ci >= 0) this.world.colliders.splice(ci, 1);
   305→   305→    this.hooks.onNoise(door.pivot.position, 11);
   306→   306→    this.hooks.netEmit('door', { id, open: true });
   307→   307→  }
   308→   308→  closeDoor(id: string) {
   309→   309→    const door = this.world.doors.get(id);
   310→   310→    if (!door || !door.open) return;
   311→   311→    door.open = false;
   312→   312→    door.pivot.rotation.y = 0;
   313→   313→    this.world.colliders.push(door.collider);
   314→   314→    this.hooks.onNoise(door.pivot.position, 9);
   315→   315→    this.hooks.netEmit('door', { id, open: false });
   316→   316→  }
   317→   317→
   318→   318→  lightBrazier(id: string) {
   319→   319→    const b = this.world.braziers.find(x => x.id === id);
   320→   320→    if (!b || b.lit) return;
   321→   321→    if (this.state.stage < 2) {
   322→   322→      this.hooks.subtitle('The coals are dead. Something must call the flame first.', 4);
   323→   323→      return;
   324→   324→    }
   325→   325→    b.lit = true;

   474→   474→  applyRemote(event: string, data: Record<string, unknown>) {
   475→   475→    switch (event) {
   476→   476→      case 'pickup': {
   477→   477→        const id = data.id as string;
   478→   478→        if (!this.state.picked.includes(id)) {
   479→   479→          this.state.picked.push(id);
   480→   480→          const mesh = this.pickupMeshes.get(id);
   481→   481→          if (mesh) mesh.visible = false;
   482→   482→          this.disableInteractable(id);
   483→   483→          const def = PICKUPS.find(p => p.id === id);
   484→   484→          if (def && NOTE_IDS.has(def.type) && !this.state.notesRead.includes(def.type)) this.state.notesRead.push(def.type);
   485→   485→        }
   486→   486→        break;
   487→   487→      }
   488→   488→      case 'door': {
   489→   489→        const door = this.world.doors.get(data.id as string);
   490→   490→        if (door) {
   491→   491→          door.open = data.open as boolean;
   492→   492→          door.pivot.rotation.y = door.open ? -Math.PI * 0.55 : 0;
   493→   493→          const ci = this.world.colliders.indexOf(door.collider);
   494→   494→          if (door.open && ci >= 0) this.world.colliders.splice(ci, 1);
   495→   495→          if (!door.open && ci < 0) this.world.colliders.push(door.collider);
   496→   496→        }
   497→   497→        break;
   498→   498→      }
   499→   499→      case 'brazier': {
   500→   500→        const b = this.world.braziers.find(x => x.id === data.id);
   501→   501→        if (b && !b.lit) {
   502→   502→          b.lit = true;
   503→   503→          if (!this.state.braziersLit.includes(b.id)) this.state.braziersLit.push(b.id);
   504→   504→          const light = new THREE.PointLight(0xff7a1a, 16, 14, 1.6);
   505→   505→          light.position.copy(b.pos3).add(new THREE.Vector3(0, 0.6, 0));
   506→   506→          this.world.root.add(light);
   507→   507→          b.light = light;
   508→   508→          if (b.flame) {
   509→   509→            b.flame.material = new THREE.MeshBasicMaterial({ color: 0xff8a2a });
   510→   510→            b.flame.scale.setScalar(1.4);
   511→   511→          }
   512→   512→        }
   513→   513→        break;
   514→   514→      }
   515→   515→      case 'stage': this.state.stage = Math.max(this.state.stage, data.stage as number); break;
   516→   516→      case 'well': {
   517→   517→        if (!this.state.picked.includes('pk_amulet') && !this.amuletSpawned) {
   518→   518→          this.amuletSpawned = true;
   519→   519→          const mesh = this.pickupMeshes.get('pk_amulet');
   520→   520→          if (mesh) mesh.visible = true;
   521→   521→          this.enableInteractable('pk_amulet');
   522→   522→        }
   523→   523→        break;
   524→   524→      }
   525→   525→      case 'altar': {
   526→   526→        if (!this.state.picked.includes('pk_iron_key')) {
   527→   527→          const mesh = this.pickupMeshes.get('pk_iron_key');
   528→   528→          if (mesh) mesh.visible = true;
   529→   529→          this.enableInteractable('pk_iron_key');
   530→   530→        }
   531→   531→        break;
   532→   532→      }
   533→   533→      case 'gate': {
   534→   534→        this.state.gateOpen = true;
   535→   535→        if (this.world.gatePivot) this.world.gatePivot.rotation.y = -Math.PI * 0.5;
   536→   536→        const gc = this.world.colliders.find(c => c.minX > -3 && c.maxX < 3 && c.minZ > 54 && c.maxZ < 56);
   537→   537→        if (gc) { const i = this.world.colliders.indexOf(gc); if (i >= 0) this.world.colliders.splice(i, 1); }
   538→   538→        break;
   539→   539→      }
   540→   540→      case 'craft': {
   541→   541→        if (data.what === 'rifle') this.state.rifleCrafted = true;
   542→   542→        if (data.what === 'remedy') this.state.doctorDone = true;
   543→   543→        break;
   544→   544→      }
   545→   545→    }
   546→   546→  }
